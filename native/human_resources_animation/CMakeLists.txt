cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(HumanResourcesAnimation LANGUAGES CXX)
  enable_testing()
endif()
add_library(human_resources_animation STATIC src/animation.cpp src/decode.cpp src/pose_math.cpp src/infantry_aim.cpp src/capture_root.cpp)
target_include_directories(human_resources_animation PUBLIC include)
target_compile_features(human_resources_animation PUBLIC cxx_std_20)
if(COMMAND game_apply_cpp_defaults)
  game_apply_cpp_defaults(human_resources_animation)
else()
  set_target_properties(human_resources_animation PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(human_resources_animation PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(human_resources_animation PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
  endif()
endif()
if(HUMAN_RESOURCES_BUILD_TESTS OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  # The parent may enable testing after adding this subdirectory. Register this
  # directory explicitly so the portable gate is present in aggregate CTest.
  enable_testing()
  add_executable(human_resources_animation_tests ../tests/animation_tests.cpp)
  target_link_libraries(human_resources_animation_tests PRIVATE human_resources_animation)
  if(COMMAND game_apply_cpp_defaults)
    game_apply_cpp_defaults(human_resources_animation_tests)
  endif()
  add_test(NAME animation_contracts COMMAND human_resources_animation_tests)
endif()
